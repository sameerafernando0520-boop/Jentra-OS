+class SceneDelegate: FlutterSceneDelegate {

+  override func sceneWillResignActive(_ scene: UIScene) {
+    super.sceneWillResignActive(scene)
+    guard let window = window, privacyCover == nil else { return }
+    let cover = UIView(frame: window.bounds)
+    cover.backgroundColor = UIColor(red: 6 / 255, green: 27 / 255, blue: 49 / 255, alpha: 1)
+    cover.autoresizingMask = [.flexibleWidth, .flexibleHeight]
+    let label = UILabel(frame: cover.bounds)
+    label.text = "Jentra Employee"
+    label.textColor = .white
+    label.textAlignment = .center
+    label.autoresizingMask = [.flexibleWidth, .flexibleHeight]
+    cover.addSubview(label)
+    window.addSubview(cover)
+    privacyCover = cover
+  }
+
+  override func sceneDidBecomeActive(_ scene: UIScene) {
+    super.sceneDidBecomeActive(scene)
+    // Flutter keeps the private view locked until online revalidation finishes.
+    privacyCover?.removeFromSuperview()
+    privacyCover = nil
+  }

+  final secureStore = NativeEmployeeKeyValueStore();

+      await Supabase.initialize(

+      setupError = const EmployeeFailure(

+        'The app could not open protected account storage. Unlock your device and restart the app.',
+      );
+    }
+  }
+  SharedPreferences? preferences;
+  try {
+    preferences = await SharedPreferences.getInstance();
+  } on Object {
+    /* Session still works without appearance preferences. */
+  }
+  final savedLanguage = preferences?.getString('employee.language');
+  final locale = Locale(
+    {'en', 'si', 'ta'}.contains(savedLanguage) ? savedLanguage! : 'en',
+  );
+  final theme = switch (preferences?.getString('employee.theme')) {
+    'light' => ThemeMode.light,
+    'dark' => ThemeMode.dark,
+    _ => ThemeMode.system,
+  };
+  final controller = EmployeeController(
+    gateway: gateway,
+    draftStore: EmployeeDraftStore(secureStore),
+  );
+  controller.error = setupError;
+  runApp(
+    EmployeeApp(
+      controller: controller,
+      initialThemeMode: theme,
+      initialLocale: locale,
+      onPreferencesChanged: (theme, locale) async {
+        await preferences?.setString('employee.theme', theme.name);
+        await preferences?.setString('employee.language', locale.languageCode);
+      },
+    ),
+  );
+  if (gateway != null) unawaited(controller.start());

+final class EmployeePendingCommand {
+  EmployeePendingCommand._(this._wire);
+  static const actions = {'link_accept', 'link_reject', 'review_acknowledge'};
+  final String _wire;
+  factory EmployeePendingCommand.create({
+    required EmployeeWorkspace workspace,
+    required String actorId,
+    required String employeeId,
+    required String action,
+    required String targetId,
+    required int expectedVersion,
+    required DateTime now,
+  }) {
+    final command = CommandEnvelope<Map<String, Object?>>(
+      commandId: const Uuid().v4(),
+      idempotencyKey: const Uuid().v4(),
+      context: workspace.context(actorId),
+      clientTimestamp: now,
+      expectedEntityVersion: expectedVersion,
+      payload: {'action': action, 'id': targetId, 'employee_id': employeeId},
+    );
+    return EmployeePendingCommand.fromJson(command.toJson((x) => x));
+  }
+  factory EmployeePendingCommand.fromJson(Map<String, Object?> json) {
+    calendarFields(json, {
+      'command_id',
+      'idempotency_key',
+      'context',
+      'client_timestamp',
+      'expected_entity_version',
+      'payload',
+    });
+    calendarUuid(json['command_id']);
+    calendarUuid(json['idempotency_key']);
+    organizationVersion(json['expected_entity_version']);
+    organizationTimestamp(json['client_timestamp']);
+    final scope = hrMap(json['context']);
+    calendarFields(scope, {
+      'tenant_id',
+      'actor_id',
+      'legal_entity_id',
+      'branch_id',
+      'warehouse_id',
+      'project_id',
+    });
+    for (final f in ['tenant_id', 'actor_id', 'legal_entity_id', 'branch_id']) {
+      calendarUuid(scope[f]);
+    }
+    if (scope['warehouse_id'] != null || scope['project_id'] != null) {
+      throw const FormatException('Unexpected scope.');
+    }
+    final payload = hrMap(json['payload']);
+    final action = payload['action'];
+    if (!actions.contains(action)) {
+      throw const FormatException('Unsupported employee action.');
+    }
+    HrCommands.validate(action as String, payload);
+    return EmployeePendingCommand._(jsonEncode(json));

+  const EmployeeConfiguration({
+    required this.url,
+    required this.publishableKey,
+  });
+  const EmployeeConfiguration.fromEnvironment()
+    : url = const String.fromEnvironment('SUPABASE_URL'),

+  final String url, publishableKey;
+  bool get isValid {
+    final parsed = Uri.tryParse(url);
+    if (parsed == null ||
+        parsed.scheme != 'https' ||
+        parsed.host.isEmpty ||
+        parsed.userInfo.isNotEmpty ||
+        parsed.hasQuery ||
+        parsed.hasFragment ||
+        (parsed.path.isNotEmpty && parsed.path != '/'))
+      return false;
+    if (publishableKey.startsWith('sb_publishable_') &&
+        RegExp(r'^sb_publishable_[A-Za-z0-9_-]{8,}$').hasMatch(publishableKey))
+      return true;
+    try {
+      final parts = publishableKey.split('.');
+      if (parts.length != 3 || parts.any((part) => part.isEmpty)) return false;
+      final payload = jsonDecode(
+        utf8.decode(base64Url.decode(base64Url.normalize(parts[1]))),
+      );
+      return payload is Map && payload['role'] == 'anon';
+    } on Object {
+      return false;
+    }
+  }

+  Future<void> _submit({
+    required String action,
+    required String targetId,
+    required String employeeId,
+    required int version,
+  }) async {
+    if (workspace == null || _actor == null) return;
+    final generation = _generation;
+    busy = true;
+    error = null;
+    notice = null;
+    _emit();
+    try {
+      final pending = EmployeePendingCommand.create(

+        actorId: _actor!,
+        employeeId: employeeId,
+        action: action,
+        targetId: targetId,
+        expectedVersion: version,
+        now: _now(),

+      // No dispatch is allowed before durable, sealed recovery exists.
+      await _draftStore.savePending(pending);
+      if (!_valid(generation)) return;
+      _pending = pending;
+      await _dispatch(pending, generation);
+    } on Object catch (e) {
+      if (_valid(generation)) error = employeeFailure(e);
+    } finally {
+      if (_valid(generation)) {
+        busy = false;
+        _emit();
+      }

+  Future<void> _dispatch(EmployeePendingCommand pending, int generation) async {
+    try {
+      await _gateway!.execute(pending);
+      if (!_valid(generation)) return;
+      await _clearPending(pending, generation);
+      if (_valid(generation)) notice = 'The server confirmed your request.';
+    } on Object catch (e) {
+      if (!_valid(generation)) return;
+      final failure = employeeFailure(e);
+      if (_accessLossCodes.contains(failure.code)) {
+        // Mask immediately. Only an explicit server rejection can delete the
+        // persisted request; authentication/session failures remain uncertain.
+        final current = workspace;
+        _lockForFailure(failure);
+        if (current != null &&
+            {
+              'PERMISSION_DENIED',
+              'SELF_LINK_REQUIRED',
+            }.contains(failure.code)) {
+          try {
+            await _draftStore.clearPending(
+              current,
+              pending.envelope.context.actorId,
+            );
+          } on Object {
+            // Keep the sealed metadata if durable removal cannot be confirmed.
+          }
+        }
+        return;
+      }
+      // Only explicit server rejection can release this command identity.
+      if ({
+        'VERSION_CONFLICT',
+        'VALIDATION_ERROR',
+        'PERMISSION_DENIED',
+        'MFA_REQUIRED',
+        'FEATURE_NOT_ENTITLED',
+        'FEATURE_NOT_INSTALLED',
+        'HR_LINK_ALREADY_EXISTS',
+        'SELF_LINK_REQUIRED',
+      }.contains(failure.code)) {
+        await _clearPending(pending, generation);
+      }
+      if (_valid(generation)) error = failure;
+    }
+  }
+
+  Future<void> _clearPending(
+    EmployeePendingCommand pending,
+    int generation,
+  ) async {
+    final c = pending.envelope.context;
+    final current = workspace;
+    if (current == null ||
+        current.key != '${c.tenantId}/${c.legalEntityId}/${c.branchId}') {
+      return;
+    }
+    await _draftStore.clearPending(current, c.actorId);
+    if (_valid(generation)) _pending = null;
+  }
+
+  Future<void> reconcileCommand() => _recover(retry: false);
+  Future<void> retryPendingCommand() => _recover(retry: true);
+  Future<void> _recover({required bool retry}) async {
+    if (!_canUsePrivateState()) return;
+    final pending = _pending;
+    if (_disposed ||
+        busy ||
+        pending == null ||
+        workspace == null ||
+        !(phase == EmployeePhase.ready ||
+            phase == EmployeePhase.linkRequired)) {
+      return;
+    }
+    final generation = _generation;
+    busy = true;
+    error = null;
+    notice = null;
+    _emit();
+    try {
+      final found = await _gateway!.hasReceipt(pending);
+      if (!_valid(generation)) return;
+      if (found) {
+        await _clearPending(pending, generation);
+        if (_valid(generation)) {
+          notice = 'The original request was confirmed by its own receipt.';
+        }
+      } else if (retry) {
+        await _dispatch(pending, generation);
+      } else {
+        notice = 'No receipt exists yet. You can explicitly retry the exact saved request.';
+      }
+    } on Object catch (e) {
+      if (_valid(generation)) {
+        final failure = employeeFailure(e);
+        if (_accessLossCodes.contains(failure.code)) {
+          _lockForFailure(failure);
+        } else {
+          error = failure;
+        }
+      }
+    } finally {
+      if (_valid(generation)) {
+        busy = false;
+        _emit();
+      }
+    }
+    if (_valid(generation) && !hasPendingCommand && error == null) {
+      await start();
+    }
+  }
+
+  Future<void> saveLeaveDraft(EmployeeLeaveDraft draft) async {
+    if (!_canUsePrivateState()) return;
+    final person = identity;
+    if (_disposed || busy || phase != EmployeePhase.ready || person == null) {
+      return;
+    }
+    final generation = _generation;
+    busy = true;
+    error = null;
+    notice = null;
+    _emit();
+    try {
+      final saved = await _draftStore.saveDraft(person, draft);
+      if (_valid(generation)) {
+        _presentDraft(saved, generation);
+        notice = 'Draft saved on this device for up to 24 hours. It has not been submitted.';
+      }
+    } on Object catch (e) {
+      if (_valid(generation)) error = employeeFailure(e);
+    } finally {
+      if (_valid(generation)) {
+        busy = false;
+        _emit();
+      }
+    }
+  }
+
+  Future<void> discardLeaveDraft() async {
+    if (!_canUsePrivateState()) return;
+    final person = identity;
+    if (_disposed || busy || person == null || phase != EmployeePhase.ready) {
+      return;
+    }
+    final generation = _generation;
+    busy = true;
+    error = null;
+    _emit();
+    try {
+      await _draftStore.discardDraft(person);
+      if (_valid(generation)) {
+        _cancelDraftRetention();
+        leaveDraft = null;
+      }
+    } on Object catch (e) {
+      if (_valid(generation)) error = employeeFailure(e);
+    } finally {
+      if (_valid(generation)) {
+        busy = false;
+        _emit();
+      }
+    }
+  }
+
+  void lock() {
+    if (_disposed || _gateway == null) return;
+    _resumeWorkspaceKey = workspace?.key ?? _resumeWorkspaceKey;
+    _generation++;
+    _mask();
+    workspace = null;
+    workspaces = const [];
+    busy = false;
+    phase = EmployeePhase.locked;
+    _emit();
+  }
+
+  Future<void> resume() async {
+    if (phase == EmployeePhase.locked) await start();
+  }
+
+  @override
+  void dispose() {
+    _disposed = true;
+    _generation++;
+    _mask();
+    unawaited(_authSubscription?.cancel());
+    super.dispose();
+  }
+}

+# Jentra Employee
+
+Android/iPhone companion for Jentra OS. This first build implements employee account access, exact account-link onboarding, shared reviews and protected local leave drafts. The complete product blueprint remains in [EMPLOYEE_APP_BLUEPRINT.md](../../docs/blueprint/mobile-apps/EMPLOYEE_APP_BLUEPRINT.md).
+
+The app reuses Jentra's supplied artwork, bundled fonts and semantic colors. Phone navigation is Home, My work, Leave, Pay and More. At enlarged text or in Sinhala/Tamil, secondary destinations remain available through More.
+
+## Supported first slice
+
+- Sign in with an existing company Jentra account.
+- Select an active company/entity/branch, resolve the accepted employee link, or review an exact HR-proposed link before accepting/rejecting it.
+- Read employee-visible shared reviews and acknowledge receipt. Receipt does not mean agreement.
+- Prepare, validate, edit and delete a local leave draft. Protected device storage, an employee/scope ownership check and a maximum 24-hour lease apply. Saving never submits leave or changes a balance.
+- Change interface language (English/Sinhala/Tamil) and light/dark/device appearance.
+- Hide private routes in the background; revalidate access on return. Sign out or lock the app.
+- Persist a sealed command before sending; check its own receipt after an uncertain outcome and retry only the original request.
+
+The development database's installed services were checked read-only on 2026-10-09. Attendance, leave balances/submission, employee payslips, employee document access, HR cases and recipient notifications still need the reviewed employee-safe backend contracts. Their screens explain availability and the next useful step. The app does not fall back to staff payroll/document permissions or show invented amounts.
+
+Accounts also holding staff review permission are still limited by the app to their resolved employee and published review states. A dedicated server projection remains a subsequent hardening step; this first version uses the existing own-link and HR APIs with app-side state filtering.
+
+## Configure and run
+
+Use the workspace SDK (Flutter 3.47.5/Dart 3.13.4 in this build). From this directory:
+
+```powershell
+flutter pub get
+flutter run --dart-define-from-file=.env.employee-development.json
+```
+
+Only the API URL and **public publishable key** belong in the client configuration. Copy `configuration.example.json` to the ignored `.env.employee-development.json` and fill the public connection fields. No service-role key or server secret is accepted. HTTPS is required.
+
+```powershell
+flutter test --no-pub --concurrency=1
+flutter analyze --no-pub
+flutter build apk --debug --no-pub --dart-define-from-file=.env.employee-development.json
+```
+
+Native secure storage is required for authenticated operation. The development web host is an unconfigured visual review surface; it never substitutes plaintext browser storage for native employee storage. Representative phone renders use fixtures under `test/`, which the production entrypoint never imports.
+
+Android debug application ID: `com.jentra.employee.dev`; the future release uses `com.jentra.employee` and requires separate signing. iOS Debug/Profile use `com.jentra.employee.dev`. Android uses a secure window, disables backup and cleartext traffic. iOS source includes a native app-switcher cover; compiling/signing and validating it require macOS/Xcode. Store distribution signing and physical-device acceptance are separate work.
+
+HR must link an existing active account to the correct employee and provide the needed existing permissions. This app never creates an employee, grants a role or infers a match from an email address. The legacy pending-link API requires `people.view`; missing access is reported to HR rather than silently broadening the role.
+
+## Verification boundaries
+
+Automated tests and rendered previews verify this source. They do not prove an authenticated employee walkthrough on Android/iPhone, installed payroll behavior, background protection on every OS, push delivery or payroll-owner acceptance. See [employee build evidence](../../docs/verification/employee-app/progress-20261009.md) for executed commands, current results and remaining gates.
